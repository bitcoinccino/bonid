Quick Start (cURL)

Verify a Citizen's BonID 
curl -X POST \
  -H "Content-Type: application/json" \
  -H "X-Partner-Api-Key: YOUR_API_KEY" \
  -H "Authorization: Bearer YOUR_TOKEN" \
  -d '{"bonid": "JM-1968-M-OUEST-P2334-217"}' \
  https://api.bonid.ht/api/v1/verify_identity
Get Full Identity Details 
curl -X GET \
  -H "X-Partner-Api-Key: YOUR_API_KEY" \
  -H "Authorization: Bearer YOUR_TOKEN" \
  https://api.bonid.ht/api/v1/identity/JM-1968-M-OUEST-P2334-217
Check Crime Status 
curl -X GET \
  -H "X-Partner-Api-Key: YOUR_API_KEY" \
  -H "Authorization: Bearer YOUR_TOKEN" \
  https://api.bonid.ht/api/v1/crime_status/JM-1968-M-OUEST-P2334-217
Exchange OAuth Token 
curl -X POST \
  -H "Content-Type: application/json" \
  -H "X-Partner-Api-Key: YOUR_API_KEY" \
  -d '{"grant_type": "authorization_code", "code": "AUTH_CODE"}' \
  https://api.bonid.ht/api/v1/oauth/token
Verify Certificate Signature (Public) 
curl -X POST \
  -H "Content-Type: application/json" \
  -d '{"certificate_id": "A3F8B2C1", "report_id": "PNH-DDO-HOMO-20260222", "algorithm": "RSASSA-PSS", "issued_at": "2026-02-22T10:30:00Z", "signature_value": "BASE64_SIG"}' \
  https://api.bonid.ht/api/v1/certificates/verify
Create Transaction Consent 
# Step 1: Request citizen consent
curl -X POST \
  -H "Content-Type: application/json" \
  -H "X-Partner-Api-Key: YOUR_API_KEY" \
  -d '{"bonid": "JM-1968-M-OUEST-P2334-217", "transaction_type": "wire_transfer", "scopes": ["profile", "address"], "amount": 5000, "currency": "HTG"}' \
  https://api.bonid.ht/api/v1/transaction_consents
Double-Lock: Get Identity with Consent 
# Step 2: After citizen approves, use consent_token to access identity data
# Only the scopes the citizen consented to will be returned
curl -X GET \
  -H "X-Partner-Api-Key: YOUR_API_KEY" \
  "https://api.bonid.ht/api/v1/identity/JM-1968-M-OUEST-P2334-217?consent_token=txn_abc123xyz"

# Without consent_token, only verification status is returned:
# { "verified": true, "bonid": "JM-1968-M-OUEST-P2334-217" }
Check Voter Eligibility (Election) 
curl -X GET \
  -H "X-BonID-Voter-Token: VOTER_SESSION_TOKEN" \
  -H "X-BonID-Attestation: APP_HMAC_NONCE" \
  -H "X-BonID-Timestamp: 1724947200" \
  -H "X-BonID-Device: DEVICE_FINGERPRINT" \
  https://api.bonid.ht/api/v1/election/2026-presidential-round1/eligibility
Cast a Vote (Election — ZKP Encrypted) 
# The server NEVER sees the vote choice — only encrypted payload
curl -X POST \
  -H "Content-Type: application/json" \
  -H "X-BonID-Voter-Token: VOTER_SESSION_TOKEN" \
  -H "X-BonID-Attestation: APP_HMAC_NONCE" \
  -H "X-BonID-Timestamp: 1724947200" \
  -H "X-BonID-Source: remote" \
  -d '{
    "nullifier": "e3b0c44298fc1c14...",
    "encrypted_choice": {
      "encrypted_choice": "BASE64...",
      "encrypted_key": "BASE64...",
      "iv": "BASE64...",
      "auth_tag": "BASE64..."
    },
    "zkp_commitment": "a8f3b2c1...",
    "zkp_proof": {
      "challenge": "HEX...",
      "response": "HEX...",
      "public_commitment": "HEX..."
    },
    "liveness_session_id": "6e183b42-a256-45b1-af0c-1b7d4c5761ae",
    "biometric_hash": "d4e5f607..."
  }' \
  https://api.bonid.ht/api/v1/election/2026-presidential-round1/cast
Verify Ballot — "Vòt ou konte" (Public) 
# No authentication needed — anyone can verify
curl -X GET \
  https://api.bonid.ht/api/v1/election/2026-presidential-round1/verify/a8f3b2c1d4e5f607...

# Response: { "verified": true, "message": "Vòt ou konte", "position": 4523 }
Register Consulate Kiosk (Election) 
curl -X POST \
  -H "Content-Type: application/json" \
  -H "X-BonID-Kiosk-Secret: CONSULATE_SECRET" \
  -d '{
    "consulate_id": "HT-CONS-MIAMI",
    "station_id": "KIOSK-01",
    "latitude": 25.7617,
    "longitude": -80.1918
  }' \
  https://api.bonid.ht/api/v1/election/2026-presidential-round1/kiosk/register
Scan Voter BonID at Kiosk (Election) 
curl -X POST \
  -H "Content-Type: application/json" \
  -H "X-BonID-Kiosk-Token: KIOSK_SESSION_TOKEN" \
  -d '{"bonid": "VP-1970-M-US-P2334-4EC"}' \
  https://api.bonid.ht/api/v1/election/2026-presidential-round1/kiosk/scan

# Response: { "status": "eligible", "voter_token": "...", "display": { "title": "Elektè Elijib" } }
